    <script>
        // Array globali per tracciare l'ordine delle lettere estratte
        let ordineStep1 = [];
        let ordineStep2 = [];
        let ordineStep3 = [];

        function mescola(array) {
            return array.sort(() => Math.random() - 0.5);
        }

        // Funzione di rendering che accetta i numeri di partenza specifici (es. 1, 3, 5)
        function renderLettere(idContainer, arrayLettere, startNum) {
            const container = document.getElementById(idContainer);
            container.innerHTML = '';
            
            const lettereMescolate = mescola([...arrayLettere]);
            
            lettereMescolate.forEach((lettera, indice) => {
                // Il numero mostrato sarà startNum + 0, startNum + 1... (quindi es. 1 e 2, oppure 3 e 4)
                const numeroAssegnato = startNum + indice;
                container.innerHTML += `
                    <div class="letter-card">
                        <div class="letter-num">${numeroAssegnato}</div>
                        <div class="letter-text">${lettera}</div>
                    </div>
                `;
            });
            return lettereMescolate;
        }

        function inizializzaGioco() {
            // Assegniamo i blocchi di numeri incrementali: 1, 3, 5 come base
            ordineStep1 = renderLettere('container-step1', ['E', 'V'], 1); // Genera numeri 1 e 2
            ordineStep2 = renderLettere('container-step2', ['O', 'I'], 3); // Genera numeri 3 e 4
            ordineStep3 = renderLettere('container-step3', ['L', 'U'], 5); // Genera numeri 5 e 6
        }

        window.onload = inizializzaGioco;

        // Trova il numero esatto abbinato a ciascuna lettera
        function calcolaSoluzioneFinale() {
            // Se la lettera è nella prima posizione del suo step prende il primo numero, altrimenti il secondo
            const numE = ordineStep1[0] === 'E' ? 1 : 2;
            const numV = ordineStep1[0] === 'V' ? 1 : 2;

            const numO = ordineStep2[0] === 'O' ? 3 : 4;
            const numI = ordineStep2[0] === 'I' ? 3 : 4;

            const numL = ordineStep3[0] === 'L' ? 5 : 6;
            const numU = ordineStep3[0] === 'U' ? 5 : 6;

            // Costruiamo la sequenza ordinata per scrivere "I LOVE U"
            const sequenzaNumerica = `${numI} - ${numL} - ${numO} - ${numV} - ${numE} - ${numU}`;

            const scatolaSoluzione = document.getElementById('soluzione-box');
            scatolaSoluzione.innerHTML = `
                <strong>Ecco il codice numerico finale:</strong><br>
                Inserisci o leggi i numeri in questo ordine per ricomporre il messaggio segreto:
                <span class="numeri-sequenza">${sequenzaNumerica}</span>
            `;
        }

        function checkCode(currentStep, correctCode, nextStepId) {
            const inputCode = document.getElementById('code' + currentStep).value.trim().toUpperCase();
            const errorMsg = document.getElementById('error' + currentStep);

            if (inputCode === correctCode) {
                errorMsg.style.display = 'none';
                document.getElementById('step' + currentStep).classList.remove('active');
                document.getElementById(nextStepId).classList.add('active');
                
                if(nextStepId === 'step-finale') {
                    calcolaSoluzioneFinale();
                }
            } else {
                errorMsg.style.display = 'block';
            }
        }
    </script>
